class TextPreprocessor:
    def __init__(self):
        self.stop_words = self.load_stop_words()
        self.astro_terms = self.load_astrology_terms()
        
    def preprocess_pipeline(self, raw_text, language="zh"):
        """文本预处理管道"""
        
        # 1. 编码规范化
        text = self.normalize_encoding(raw_text)
        
        # 2. 格式清理
        text = self.clean_formatting(text)
        
        # 3. 特殊字符处理
        text = self.handle_special_characters(text)
        
        # 4. 占星术语标准化
        text = self.standardize_astro_terms(text)
        
        # 5. 语言特定处理
        if language == "zh":
            text = self.process_chinese_text(text)
        elif language == "en":
            text = self.process_english_text(text)
            
        # 6. 质量验证
        if not self.quality_validation(text):
            raise ValueError("文本质量不达标")
            
        return text
    
    def clean_formatting(self, text):
        """清理格式字符"""
        import re
        
        # 移除HTML标签
        text = re.sub(r'<[^>]+>', '', text)
        
        # 移除多余空白
        text = re.sub(r'\s+', ' ', text)
        
        # 移除特殊格式字符
        text = re.sub(r'[^\w\s\u4e00-\u9fff.,!?;:\-()（）、。，！？；：]', '', text)
        
        # 统一标点符号
        text = text.replace('。', '. ').replace('，', ', ').replace('！', '! ').replace('？', '? ')
        
        return text.strip()
    
    def standardize_astro_terms(self, text):
        """占星术语标准化"""

class SmartChunker:
    def __init__(self, max_chunk_size=512, overlap_size=50):
        self.max_chunk_size = max_chunk_size
        self.overlap_size = overlap_size
        self.sentence_splitter = self.init_sentence_splitter()
    
    def chunk_document(self, text, metadata=None):
        """智能文档分块"""
        
        # 1. 语义边界检测
        semantic_boundaries = self.detect_semantic_boundaries(text)
        
        # 2. 基于语义的分块
        chunks = self.semantic_chunking(text, semantic_boundaries)
        
        # 3. 大小优化
        optimized_chunks = self.optimize_chunk_sizes(chunks)
        
        # 4. 添加重叠内容
        overlapped_chunks = self.add_overlap(optimized_chunks)
        
        # 5. 生成元数据
        chunk_objects = []
        for i, chunk in enumerate(overlapped_chunks):
            chunk_metadata = {
                "chunk_id": f"chunk_{i}",
                "chunk_index": i,
                "total_chunks": len(overlapped_chunks),
                "char_count": len(chunk),
                "word_count": len(chunk.split()),
                "topics": self.extract_topics(chunk),
                "astro_entities": self.extract_astro_entities(chunk),
                "semantic_similarity": self.calculate_similarity_with_neighbors(chunk, overlapped_chunks, i)
            }
            
            if metadata:
                chunk_metadata.update(metadata)
            
            chunk_objects.append({
                "content": chunk,
                "metadata": chunk_metadata
            })
        
        return chunk_objects
    
    def detect_semantic_boundaries(self, text):
        """检测语义边界"""
        import spacy
        
        # 加载中文NLP模型
        nlp = spacy.load("zh_core_web_sm")
        doc = nlp(text)
        
        boundaries = []
        current_topic = None
        
        for sent in doc.sents:
            # 主题检测
            sent_topics = self.extract_sentence_topics(sent.text)
            
            if current_topic is None:
                current_topic = sent_topics
            elif not self.topics_similar(current_topic, sent_topics):
                # 主题变化，标记为边界点
                boundaries.append(sent.start_char)
                current_topic = sent_topics
        
        return boundaries
    
    def extract_astro_entities(self, text):
        """提取占星实体"""
        entities = {
            "zodiac_signs": [],
            "planets": [],
            "houses": [],
            "aspects": [],

class EmbeddingProcessor:
    def __init__(self, model_name="BAAI/bge-m3"):
        self.model = SentenceTransformer(model_name)
        self.dimension = 1024  # BGE-M3模型维度
        self.batch_size = 32
        
    def generate_embeddings(self, texts, show_progress=True):
        """生成文本嵌入向量"""
        
        if show_progress:
            print(f"开始向量化{len(texts)}个文本块...")
        
        # 批量处理
        embeddings = []
        for i in tqdm(range(0, len(texts), self.batch_size)):
            batch = texts[i:i+self.batch_size]
            
            # 预处理批次文本
            processed_batch = [self.preprocess_for_embedding(text) for text in batch]
            
            # 生成嵌入向量
            batch_embeddings = self.model.encode(
                processed_batch,
                normalize_embeddings=True,  # 归一化向量
                show_progress_bar=False
            )
            
            embeddings.extend(batch_embeddings)
        
        return np.array(embeddings)
    
    def preprocess_for_embedding(self, text):
        """嵌入前的文本预处理"""
        
        # 1. 长度限制（避免超出模型最大长度）
        max_length = 512
        if len(text) > max_length:

```python
class IntelligentRetriever:
    def __init__(self, milvus_manager, embedding_processor):
        self.milvus = milvus_manager
        self.embedder = embedding_processor
        self.collection = Collection(milvus_manager.collection_name)
        
    def search(self, query, top_k=5, filters=None):
        """智能检索"""
        
        # 1. 查询预处理
        processed_query = self.preprocess_query(query)
        
        # 2. 生成查询向量
        query_embedding = self.embedder.generate_embeddings([processed_query])[0]
        
        # 3. 多策略检索
        results = self.multi_strategy_search(query_embedding, top_k, filters)
        
        # 4. 结果后处理
        processed_results = self.postprocess_results(results, query)
        
        return processed_results
    
    def preprocess_query(self, query):
        """查询预处理"""
        
        # 1. 查询扩展
        expanded_query = self.expand_query(query)
        
        # 2. 术语标准化
        standardized_query = self.standardize_query_terms(expanded_query)
        
        # 3. 关键词增强
        enhanced_query = self.enhance_query_keywords(standardized_query)
        
        return enhanced_query
    
    def expand_query(self, query):
        """查询扩展"""
        
        # 同义词扩展
        synonym_map = {
            "运势": ["运势", "运程", "运气", "命运"],
            "性格": ["性格", "个性", "特征", "特质"],
            "爱情": ["爱情", "感情", "恋爱", "情感"],
            "事业": ["事业", "工作", "职业", "事业运"],
            "财运": ["财运", "金钱", "财富", "财务"]
        }
        
        expanded_terms = []
        for word in query.split():
            if word in synonym_map:
                expanded_terms.extend(synonym_map[word])
            else:
                expanded_terms.append(word)
        
        return " ".join(expanded_terms)
    
    def multi_strategy_search(self, query_embedding, top_k, filters):
        """多策略检索"""
        
        # 策略1: 向量相似度检索
        vector_results = self.vector_similarity_search(query_embedding, top_k*2)
        
        # 策略2: 混合检索（向量+关键词）
        hybrid_results = self.hybrid_search(query_embedding, top_k*2, filters)
        
        # 策略3: 语义检索
        semantic_results = self.semantic_search(query_embedding, top_k*2)
        
        # 结果融合
        merged_results = self.merge_search_results([
            vector_results, hybrid_results, semantic_results
        ], top_k)
        
        return merged_results
    
    def vector_similarity_search(self, query_embedding, top_k):
        """向量相似度检索"""
        
        search_params = {
            "metric_type": "COSINE",
            "params": {"nprobe": 64}
        }
        
        results = self.collection.search(
            data=[query_embedding.tolist()],
            anns_field="embedding",
            param=search_params,
            limit=top_k,
            output_fields=["content", "source", "category", "priority", "metadata_json"]
        )
        
        return results[0]  # 返回第一个查询的结果
    
    def merge_search_results(self, result_lists, final_top_k):
        """结果融合算法"""
        
        # 使用RRF (Reciprocal Rank Fusion)算法
        score_dict = {}
        
        for i, results in enumerate(result_lists):
            for rank, result in enumerate(results):
                doc_id = result.id
                
                # RRF得分计算
                rrf_score = 1.0 / (rank + 60)  # 60是常用的k值
                
                if doc_id not in score_dict:
                    score_dict[doc_id] = {
                        "score": 0,
                        "result": result,
                        "strategy_scores": {}
                    }
                
                score_dict[doc_id]["score"] += rrf_score
                score_dict[doc_id]["strategy_scores"][f"strategy_{i}"] = rrf_score
        
        # 按总分排序
        sorted_results = sorted(
            score_dict.values(),
            key=lambda x: x["score"],
            reverse=True
        )
        
        return [item["result"] for item in sorted_results[:final_top_k]]
```

        # 判断整体健康状态
        if any(status != "healthy" for status in health_status.values() if isinstance(status, str)):
            health_status["overall"] = "unhealthy"
        
        return health_status
    
    def auto_maintenance(self):
        """自动维护任务"""
        
        print("开始执行自动维护任务...")
        
        # 1. 清理过期缓存
        self.cleanup_expired_cache()
        
        # 2. 压缩向量索引
        self.compact_vector_index()
        
        # 3. 更新统计信息
        self.update_statistics()
        
        # 4. 检查数据一致性
        consistency_report = self.check_data_consistency()
        
        print("自动维护任务完成")
        return consistency_report
